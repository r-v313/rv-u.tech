❌ VULNERABLE LOGIC (UNAUTHORIZED ACCESS) app.get('/api/v1/invoices/:id', async (req, res) => {
  const invoiceId = req.params.id;
  // MISSING: Ownership verification of invoiceId against req.user.id
  const invoice = await Invoice.findById(invoiceId);
  return res.json(invoice);
});
🟢 REMEDIATED LOGIC (SECURE AUTHZ) app.get('/api/v1/invoices/:id', async (req, res) => {
  const invoiceId = req.params.id;
  // SECURE: Verify object ownership against authenticated session
  const invoice = await Invoice.findOne({ _id: invoiceId, userId: req.user.id });
  if (!invoice) return res.status(403).json({ error: "Access Denied" });
  return res.json(invoice);
});
